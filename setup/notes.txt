## GRUB
# reduce grub timeout
sudo nano /etc/default/grub
GRUB_TIMEOUT = 2
sudo grub-mkconfig -o /boot/grub/grub.cfg

# set grub resolution
sudo nano /etc/default/grub
GRUB_GFXMODE=1920x1080
GRUB_GFXPAYLOAD_LINUX="keep"
sudo grub-mkconfig -o /boot/grub/grub.cfg

## POWER OPTIONS
# allow user to trigger poweroptions without needing to enter a password
sudo groupadd npwpower
sudo usermod $(id -nu 1000) -aG !$

sudo visudo
# add the line, where the sbin commands are what we want to use to shutdown/reboot/{poweroption}
npwpower ALL=(ALL) NOPASSWD: /sbin/poweroff, /sbin/reboot
