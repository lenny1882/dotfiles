## APPS
# set apps as default
# vivaldi-stable
~/.config/mimeapps.list
	>	[Default Applications]
		text/html=vivaldi-stable.desktop
		x-scheme-handler/http=vivaldi-stable.desktop
		x-scheme-handler/https=vivaldi-stable.desktop
		x-scheme-handler/about=vivaldi-stable.desktop
		x-scheme-handler/unknown=vivaldi-stable.desktop
		x-scheme-handler/rambox=rambox.desktop
		x-scheme-handler/claude-cli=claude-code-url-handler.desktop
	>	[Added Associations]  # empty
# everything else (inode/directory=nnn.desktop, images, video, text) comes from the .desktop files themselves
# $BROWSER=vivaldi-stable is set in shellrc/.shellrc_env

> may need to: rm ~/.local/share/applications/mimeapps.list && ln -s ~/.config/mimeapps.list ~/.local/share/applications/mimeapps.list
	> .local copy is legacy, should now be in .config -- if .local version contains things, copy it
> if making changes, add to two sections: [Default Applications], [Added Associations]

$ xdg-mime query default application/pdf
$ xdg-mime default vivaldi-stable.desktop application/pdf
